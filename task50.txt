class Solution {
int count = 0;
int result = 0;

public int kthSmallest(TreeNode root, int k) {
inorder(root, k);
return result;
}

private void inorder(TreeNode root, int k) {

if (root == null) {
return;
}

// Visit left subtree
inorder(root.left, k);

// Count current node
count++;

if (count == k) {
result = root.val;
return;
}

// Visit right subtree
inorder(root.right, k);
}
}
