
import java.util.*;

class Solution {

public List<List<Integer>> verticalTraversal(TreeNode root) {

// Store column -> row -> node values
TreeMap<Integer, TreeMap<Integer, PriorityQueue<Integer>>> map =
new TreeMap<>();

Queue<Object[]> queue = new LinkedList<>();

queue.offer(new Object[]{root, 0, 0});

while (!queue.isEmpty()) {

Object[] data = queue.poll();

TreeNode node = (TreeNode) data[0];
int row = (int) data[1];
int col = (int) data[2];

map.putIfAbsent(col, new TreeMap<>());

map.get(col).putIfAbsent(row, new PriorityQueue<>());

map.get(col).get(row).offer(node.val);

// Left child
if (node.left != null) {
queue.offer(new Object[]{node.left, row + 1, col - 1});
}

// Right child
if (node.right != null) {
queue.offer(new Object[]{node.right, row + 1, col + 1});
}
}

List<List<Integer>> result = new ArrayList<>();

// Process columns from left to right
for (TreeMap<Integer, PriorityQueue<Integer>> rows : map.values()) {

List<Integer> column = new ArrayList<>();

// Process rows from top to bottom
for (PriorityQueue<Integer> nodes : rows.values()) {

while (!nodes.isEmpty()) {
column.add(nodes.poll());
}
}

result.add(column);
}

return result;
}
}
